## Technical Deep Dive
RLHF involves training a model to maximize a reward function that is learned from human feedback. The process typically involves the following steps:

1. Collecting human feedback on model outputs
2. Training a reward model to predict human preferences
3. Using reinforcement learning to optimize the model based on the reward model

Let's dive into a complete, runnable code example using PyTorch and the Transformers library.

### Step 2: Training a Reward Model
Next, we'll train a reward model to predict human preferences.

```python
import torch.nn as nn
import torch.optim as optim

# Define a custom dataset class for our human feedback data
class HumanFeedbackDataset(torch.utils.data.Dataset):
    def __init__(self, input_texts, generated_texts, feedback):
        self.input_texts = input_texts
        self.generated_texts = generated_texts
        self.feedback = feedback

    def __getitem__(self, idx):
        input_text = self.input_texts[idx]
        generated_text = self.generated_texts[idx]
        feedback = self.feedback[idx]

        encoding = tokenizer(input_text, generated_text, return_tensors="pt")
        return {
            "input_ids": encoding["input_ids"].flatten(),
            "attention_mask": encoding["attention_mask"].flatten(),
            "labels": torch.tensor(feedback, dtype=torch.float),
        }

    def __len__(self):
        return len(self.input_texts)

# Create a dataset and data loader
dataset = HumanFeedbackDataset(["input_text1", "input_text2"], ["generated_text1", "generated_text2"], [0.5, 0.8])
data_loader = torch.utils.data.DataLoader(dataset, batch_size=32, shuffle=True)

# Train the reward model
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
reward_model.to(device)
criterion = nn.MSELoss()
optimizer = optim.Adam(reward_model.parameters(), lr=1e-5)

for epoch in range(5):
    for batch in data_loader:
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)
        labels = batch["labels"].to(device)

        optimizer.zero_grad()

        outputs = reward_model(input_ids, attention_mask=attention_mask, labels=labels)
        loss = criterion(outputs.logits, labels)

        loss.backward()
        optimizer.step()

    print(f"Epoch {epoch+1}, Loss: {loss.item():.4f}")
```
